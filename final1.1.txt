<script>
    let canvas, ctx;
    let vBuffer = []; let iBuffer = [];
    
    let isTestActive = false, isEStopActive = false, isTestComplete = false, hasTripped = false;
    let animationId, tick = 0;
    let currentTestType = null, waveColorI = '#00ff66', waveColorV = '#3b82f6';

    let fixedScaleV = 1, fixedScaleI = 1;

    // --- AUDIO ENGINE ---
    let audioCtx = null;
    let humOscillator = null;
    let alarmInterval = null;
    let audioEnabled = false;

    function toggleAudio() {
        if (!audioCtx) {
            audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        }
        audioEnabled = !audioEnabled;
        const btn = document.getElementById('audio-toggle');
        if (audioEnabled) {
            if(audioCtx.state === 'suspended') audioCtx.resume();
            btn.innerText = "AUDIO: ON";
            btn.style.color = "#00ff66";
            btn.style.borderColor = "#047857";
        } else {
            btn.innerText = "AUDIO: OFF (CLICK TO ENABLE)";
            btn.style.color = "#64748b";
            btn.style.borderColor = "#1a3324";
            stopHum();
            stopAlarm();
        }
    }

    function playRelayClick() {
        if (!audioEnabled || !audioCtx) return;
        const osc = audioCtx.createOscillator();
        const gainNode = audioCtx.createGain();
        osc.type = 'square';
        osc.frequency.setValueAtTime(150, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(40, audioCtx.currentTime + 0.05);
        gainNode.gain.setValueAtTime(1, audioCtx.currentTime);
        gainNode.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.05);
        osc.connect(gainNode);
        gainNode.connect(audioCtx.destination);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.05);
    }

    function playArcSnap() {
        if (!audioEnabled || !audioCtx) return;
        const bufferSize = audioCtx.sampleRate * 0.1; 
        const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
        const data = buffer.getChannelData(0);
        for (let i = 0; i < bufferSize; i++) data[i] = Math.random() * 2 - 1;
        const noise = audioCtx.createBufferSource();
        noise.buffer = buffer;
        const gainNode = audioCtx.createGain();
        gainNode.gain.setValueAtTime(1, audioCtx.currentTime);
        gainNode.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.1);
        noise.connect(gainNode);
        gainNode.connect(audioCtx.destination);
        noise.start();
    }

    function startHum() {
        if (!audioEnabled || !audioCtx) return;
        if (humOscillator) humOscillator.stop();
        humOscillator = audioCtx.createOscillator();
        const gainNode = audioCtx.createGain();
        humOscillator.type = 'sine';
        humOscillator.frequency.value = 50; 
        gainNode.gain.value = 0.1; // Keep it subtle
        humOscillator.connect(gainNode);
        gainNode.connect(audioCtx.destination);
        humOscillator.start();
    }

    function stopHum() {
        if (humOscillator) { humOscillator.stop(); humOscillator = null; }
    }

    function playAlarmBeep() {
        if (!audioEnabled || !audioCtx) return;
        const osc = audioCtx.createOscillator();
        const gainNode = audioCtx.createGain();
        osc.type = 'sawtooth';
        osc.frequency.value = 800;
        gainNode.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gainNode.gain.setValueAtTime(0, audioCtx.currentTime + 0.2);
        osc.connect(gainNode);
        gainNode.connect(audioCtx.destination);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.2);
    }

    function startAlarm() {
        if (alarmInterval) clearInterval(alarmInterval);
        playAlarmBeep();
        alarmInterval = setInterval(playAlarmBeep, 500);
    }

    function stopAlarm() {
        if (alarmInterval) { clearInterval(alarmInterval); alarmInterval = null; }
    }
    // ----------------------

    window.onload = function() {
        canvas = document.getElementById('waveformCanvas');
        ctx = canvas.getContext('2d');
        initCanvas();
        generatePoles(); 
        const rect = canvas.getBoundingClientRect();
        for(let i=0; i<rect.width; i++) { vBuffer.push(0); iBuffer.push(0); }
        animationId = requestAnimationFrame(renderLoop);
        logMessage('[SYSTEM] Local HMI rendering engine initialized.', 'new');
    };

    function initCanvas() {
        if (!canvas) return;
        const dpr = window.devicePixelRatio || 1;
        const rect = canvas.parentElement.getBoundingClientRect();
        canvas.width = rect.width * dpr; canvas.height = rect.height * dpr;
        ctx.scale(dpr, dpr); canvas.style.width = `${rect.width}px`; canvas.style.height = `${rect.height}px`;
    }

    window.onresize = initCanvas;

    function generatePoles() {
        const count = parseInt(document.getElementById('poles').value);
        const container = document.getElementById('mcb-switches');
        targetVoltageRMS = (count >= 3) ? 400 : 230;
        container.innerHTML = '';
        for(let i = 0; i < count; i++) {
            container.innerHTML += `
                <div class="mcb-body"><div class="mcb-terminal top"></div><div class="mcb-led"></div>
                <div class="mcb-switch-bay"><div class="mcb-spark"></div><div class="mcb-lever off"></div></div>
                <div class="mcb-terminal bottom"></div></div>`;
        }
    }

    function logMessage(msg, type = 'new') {
        const box = document.getElementById('logBox');
        if(box) {
            const time = new Date().toLocaleTimeString('en-US', { hour12: false, fractionalSecondDigits: 2 });
            const div = document.createElement('div'); div.className = `log-${type}`; div.innerText = `[${time}] ${msg}`;
            box.appendChild(div); box.scrollTop = box.scrollHeight;
        }
    }

    function setControls(disabled) {
        document.getElementById('btn-start').disabled = disabled; document.getElementById('btn-fault').disabled = disabled;
        document.getElementById('poles').disabled = disabled; document.getElementById('rating').disabled = disabled;
        document.getElementById('temp').disabled = disabled;
    }

    function updateMCBState(state, text) {
        const levers = document.querySelectorAll('.mcb-lever'), leds = document.querySelectorAll('.mcb-led');
        const sparks = document.querySelectorAll('.mcb-spark'), hardwareContainer = document.getElementById('mcb-switches');
        
        document.getElementById('mcb-label').innerHTML = text;
        hardwareContainer.className = 'mcb-switches-container'; 
        levers.forEach(l => l.className = 'mcb-lever'); leds.forEach(l => l.className = 'mcb-led');

        if (state === 'testing') {
            levers.forEach(l => l.classList.add('on')); hardwareContainer.classList.add('testing'); leds.forEach(l => l.classList.add('on'));
        } else if (state === 'tripped') {
            levers.forEach(l => l.classList.add('tripped'));
            sparks.forEach(s => { s.classList.remove('flash'); void s.offsetWidth; s.classList.add('flash'); });
        } else if (state === 'welded') {
            levers.forEach(l => l.classList.add('on')); leds.forEach(l => l.classList.add('fault'));
        } else {
            levers.forEach(l => l.classList.add('off'));
        }
    }

    function renderLoop() { generateDataPoint(); drawCanvas(); animationId = requestAnimationFrame(renderLoop); }

    function tripHardware(isNormalPass) {
        hasTripped = true;
        stopHum();
        
        const status = document.getElementById('status');
        if (isNormalPass) {
            playArcSnap();
            playRelayClick();
            status.className = 'status-bar pass'; status.innerText = 'TEST COMPLETE : PASS';
            updateMCBState('tripped', 'TRIPPED<br>(OPEN)');
            logMessage(`MCB Tripped successfully. Peak RMS Current: ${(maxI/Math.SQRT2).toFixed(2)}A`, 'ok');
        } else {
            playRelayClick(); // Backup contactor opening
            status.className = 'status-bar fail'; status.innerText = 'BACKUP ISOLATION ENGAGED';
            updateMCBState('welded', 'STUCK!<br>(FAULT)');
            logMessage('MCB Failed to trip. Safety contactor opened circuit.', 'err');
        }
    }

    function freezeGraph() { isTestActive = false; isTestComplete = true; setControls(false); document.getElementById('btn-export').disabled = false; }

    function generateDataPoint() {
        if (isTestComplete) return; 

        let valV = 0, valI = 0; tick++;
        const peakV = targetVoltageRMS * Math.SQRT2, angularFreq = 0.15, dt = 0.000478; 

        if (isEStopActive) {
            valV = 0; valI = 0;
        } else if (isTestActive) {
            const currentRating = parseFloat(document.getElementById('rating').value);
            
            if (currentTestType === 'fault') {
                if (tick < 10) { valV = peakV * Math.sin(tick * angularFreq); valI = 0;
                } else if (tick < faultSpikeTicks) {
                    valV = (peakV * 0.1) * Math.sin(tick * angularFreq); 
                    valI = (currentRating * 80) * Math.sin(tick * angularFreq) * Math.exp(-(tick-10)*0.01) + (currentRating * 30);
                } else {
                    if (!hasTripped) tripHardware(false); 
                    valV = peakV * Math.sin(tick * angularFreq); valI = iBuffer[iBuffer.length-1] * 0.6; 
                }
            } else {
                if (tick < 10) { valV = peakV * Math.sin(tick * angularFreq); valI = 0;
                } else if (tick < tripDelayTicks) {
                    valV = (peakV * 0.95) * Math.sin(tick * angularFreq);
                    valI = (currentRating * 3.5) * Math.sin(tick * angularFreq - 0.5); 
                } else {
                    if (!hasTripped) tripHardware(true); 
                    valV = peakV * Math.sin(tick * angularFreq); valI = iBuffer[iBuffer.length-1] * 0.6; 
                }
            }

            const canvasHeight = canvas.getBoundingClientRect().height;
            if (tick === 11) {
                let expectedMaxV = peakV * 1.2;
                let expectedMaxI = currentTestType === 'fault' ? (currentRating * 110) : (currentRating * 5);
                fixedScaleV = (canvasHeight / 2) / expectedMaxV; fixedScaleI = (canvasHeight / 2) / expectedMaxI;
            }

            if (Math.abs(valI) > maxI) maxI = Math.abs(valI); if (Math.abs(valV) > maxV) maxV = Math.abs(valV);
            const instantPower = Math.abs((valV * valI) / 1000); if (instantPower > maxP) maxP = instantPower;
            if (Math.abs(valI) > 5) totalI2t += Math.pow(valI, 2) * dt; 

            document.getElementById('readout-v').innerText = (maxV / Math.SQRT2).toFixed(0);
            document.getElementById('readout-i').innerText = (maxI / Math.SQRT2).toFixed(2);
            document.getElementById('readout-p').innerText = maxP.toFixed(2);
            document.getElementById('readout-i2t').innerText = totalI2t.toFixed(0);

            const endTick = currentTestType === 'fault' ? faultSpikeTicks + 50 : tripDelayTicks + 50;
            if (tick > endTick) freezeGraph(); 
            
        } else {
            valV = peakV * Math.sin(tick * angularFreq); valI = (Math.random() - 0.5) * 1.5; 
            document.getElementById('readout-v').innerText = targetVoltageRMS;
            const canvasHeight = canvas.getBoundingClientRect().height;
            fixedScaleV = (canvasHeight / 2) / (peakV * 1.2); fixedScaleI = (canvasHeight / 2) / 20;
        }

        vBuffer.push(valV); iBuffer.push(valI);
        const rect = canvas.getBoundingClientRect();
        if (vBuffer.length > rect.width) vBuffer.shift(); if (iBuffer.length > rect.width) iBuffer.shift();
    }

    function drawCanvas() {
        const rect = canvas.getBoundingClientRect();
        ctx.clearRect(0, 0, rect.width, rect.height);
        
        ctx.strokeStyle = 'rgba(10, 31, 16, 0.5)'; ctx.lineWidth = 1; ctx.beginPath();
        for(let i = 0; i < rect.width; i += 40) { ctx.moveTo(i, 0); ctx.lineTo(i, rect.height); }
        for(let i = 0; i < rect.height; i += 40) { ctx.moveTo(0, i); ctx.lineTo(rect.width, i); }
        ctx.stroke();
        
        ctx.strokeStyle = '#164024'; ctx.beginPath(); ctx.moveTo(0, rect.height/2); ctx.lineTo(rect.width, rect.height/2); ctx.stroke();
        ctx.strokeStyle = 'rgba(59, 130, 246, 0.6)'; ctx.lineWidth = 1.5; ctx.shadowBlur = 4; ctx.shadowColor = waveColorV; ctx.beginPath();
        for (let i = 0; i < vBuffer.length; i++) {
            const y = (rect.height / 2) - (vBuffer[i] * fixedScaleV);
            if (i === 0) ctx.moveTo(i, y); else ctx.lineTo(i, y);
        }
        ctx.stroke();

        ctx.strokeStyle = waveColorI; ctx.lineWidth = 2.5; ctx.shadowBlur = 10; ctx.shadowColor = waveColorI; ctx.beginPath();
        for (let i = 0; i < iBuffer.length; i++) {
            const y = (rect.height / 2) - (iBuffer[i] * fixedScaleI);
            if (i === 0) ctx.moveTo(i, y); else ctx.lineTo(i, y);
        }
        ctx.stroke(); ctx.shadowBlur = 0;
    }

    function runTest(isFault) {
        if (isTestActive || isEStopActive) return;
        
        const poles = document.getElementById('poles').value;
        const rating = parseFloat(document.getElementById('rating').value);
        const temp = parseInt(document.getElementById('temp').value);

        isTestActive = true; isTestComplete = false; hasTripped = false; document.getElementById('btn-export').disabled = true; 

        currentTestType = isFault ? 'fault' : 'normal';
        waveColorI = isFault ? '#ff3333' : '#00ff66';
        tick = 0; maxI = 0; maxV = 0; maxP = 0; totalI2t = 0;

        // Base timings
        if (rating === 6) { tripDelayTicks = 60; faultSpikeTicks = 90; 
        } else if (rating === 10 || rating === 16) { tripDelayTicks = 100; faultSpikeTicks = 120; 
        } else if (rating === 32) { tripDelayTicks = 150; faultSpikeTicks = 170; }
        
        // Apply Temperature Derating (Hotter = Faster Trip)
        const tempDerating = 1 - ((temp - 30) * 0.015);
        tripDelayTicks = Math.max(20, tripDelayTicks * tempDerating);
        
        setControls(true);
        updateMCBState('testing', 'LIVE<br>(CLOSED)');
        
        playRelayClick();
        startHum();

        const hardwareContainer = document.getElementById('mcb-switches');
        if (rating >= 16) hardwareContainer.classList.add('heavy-load');
        
        const status = document.getElementById('status');
        status.className = 'status-bar testing';
        status.innerText = isFault ? 'INJECTING FAULT SCENARIO' : 'EXECUTING IEC TRIP PROTOCOL';
        
        logMessage(`Starting simulation: ${poles}P MCB (${rating}A). Temp adjusted: ${temp}°C`, 'new');
    }

    function toggleEStop() { if (isEStopActive) resetSystem(); else engageEStop(); }

    function engageEStop() {
        isTestActive = false; isEStopActive = true; isTestComplete = false; setControls(true);
        document.getElementById('btn-estop').disabled = false; document.getElementById('btn-export').disabled = true;
        
        stopHum();
        playRelayClick();
        startAlarm();

        waveColorI = '#ff3333';
        for(let i = 0; i < vBuffer.length; i++) { vBuffer[i] = 0; iBuffer[i] = 0; }

        document.getElementById('main-container').classList.add('emergency-mode');
        const estopBtn = document.getElementById('btn-estop');
        estopBtn.innerText = '⚠ RESET SYSTEM ⚠'; estopBtn.classList.add('btn-reset');

        updateMCBState('estop', 'LOCKED<br>(OPEN)');
        const status = document.getElementById('status');
        status.className = 'status-bar fail'; status.innerText = 'SYSTEM LOCKED: HARDWARE E-STOP ENGAGED';
        logMessage('CRITICAL: Operator triggered Emergency Stop. All relays opened.', 'err');
    }

    function resetSystem() {
        isEStopActive = false; isTestComplete = false; hasTripped = false; setControls(false);
        waveColorI = '#00ff66';
        
        stopAlarm();
        playRelayClick();

        document.getElementById('main-container').classList.remove('emergency-mode');
        const estopBtn = document.getElementById('btn-estop');
        estopBtn.innerText = 'E-STOP'; estopBtn.classList.remove('btn-reset');
        updateMCBState('idle', 'IDLE<br>(OPEN)');

        const status = document.getElementById('status');
        status.className = 'status-bar'; status.innerText = 'SYSTEM IDLE : READY FOR TEST';
        logMessage('Safety interlock reset. Relays energized and ready.', 'ok');
        
        document.getElementById('readout-i').innerText = '0.00'; document.getElementById('readout-p').innerText = '0.00'; document.getElementById('readout-i2t').innerText = '0.00';
    }

    function exportCSV() {
        let csvContent = "data:text/csv;charset=utf-8,Index,Voltage(V),Current(A)\n";
        vBuffer.forEach((v, index) => { csvContent += `${index},${v.toFixed(2)},${iBuffer[index].toFixed(4)}\n`; });
        const encodedUri = encodeURI(csvContent);
        const link = document.createElement("a"); link.setAttribute("href", encodedUri);
        link.setAttribute("download", `mcb_telemetry_${Date.now()}.csv`);
        document.body.appendChild(link); link.click(); document.body.removeChild(link);
        logMessage('Telemetry data successfully exported to CSV.', 'ok');
    }
