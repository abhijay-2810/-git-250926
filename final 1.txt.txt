<script>
    let canvas, ctx;
    let dataBuffer = [];
    let isTestActive = false;
    let isEStopActive = false;
    let animationId;
    let tick = 0;
    
    let currentTestType = null;
    let waveColor = '#00ff66';
    let maxIp = 0;
    let totalI2t = 0;

    window.onload = function() {
        canvas = document.getElementById('waveformCanvas');
        ctx = canvas.getContext('2d');
        initCanvas();
        
        const rect = canvas.getBoundingClientRect();
        for(let i=0; i<rect.width; i++) {
            dataBuffer.push(0);
        }
        
        animationId = requestAnimationFrame(renderLoop);
        logMessage('[SYSTEM] Local HMI rendering engine initialized.', 'new');
    };

    function initCanvas() {
        if (!canvas) return;
        const dpr = window.devicePixelRatio || 1;
        const rect = canvas.parentElement.getBoundingClientRect();
        
        canvas.width = rect.width * dpr;
        canvas.height = 260 * dpr;
        ctx.scale(dpr, dpr);
        canvas.style.width = `${rect.width}px`;
        canvas.style.height = '260px';
    }

    window.onresize = initCanvas;

    function logMessage(msg, type = 'new') {
        const box = document.getElementById('logBox');
        if(box) {
            const time = new Date().toLocaleTimeString('en-US', { hour12: false, fractionalSecondDigits: 2 });
            const div = document.createElement('div');
            div.className = `log-${type}`;
            div.innerText = `[${time}] ${msg}`;
            box.appendChild(div);
            box.scrollTop = box.scrollHeight;
        }
    }

    function setControls(disabled) {
        document.getElementById('btn-start').disabled = disabled;
        document.getElementById('btn-fault').disabled = disabled;
        document.getElementById('poles').disabled = disabled;
        document.getElementById('rating').disabled = disabled;
    }

    function renderLoop() {
        generateDataPoint();
        drawCanvas();
        animationId = requestAnimationFrame(renderLoop);
    }

    function generateDataPoint() {
        let val = 0;

        if (isEStopActive) {
            val = 0; 
        } else if (isTestActive) {
            tick++;
            
            if (currentTestType === 'fault') {
                if (tick < 10) val = 0;
                else if (tick < 120) val = Math.sin((tick - 10) * 0.1) * 150; 
                else val = (Math.random() - 0.5) * 2; 
            } else {
                if (tick < 10) val = 0;
                else if (tick < 80) val = Math.sin((tick - 10) * 0.2) * 100 * Math.exp(-(tick - 10) * 0.03);
                else val = (Math.random() - 0.5) * 2; 
            }

            if (Math.abs(val) > maxIp) maxIp = Math.abs(val);
            if (Math.abs(val) > 5) totalI2t += Math.pow(val, 2) * 0.01; 

            document.getElementById('readout-ip').innerText = (maxIp / 10).toFixed(2);
            document.getElementById('readout-i2t').innerText = totalI2t.toFixed(0);

            if (tick > 200) finishTest();
            
        } else {
            val = (Math.random() - 0.5) * 1.5;
        }

        dataBuffer.push(val);
        const rect = canvas.getBoundingClientRect();
        if (dataBuffer.length > rect.width) {
            dataBuffer.shift();
        }
    }

    function drawCanvas() {
        const rect = canvas.getBoundingClientRect();
        ctx.clearRect(0, 0, rect.width, rect.height);
        
        ctx.strokeStyle = 'rgba(10, 31, 16, 0.5)';
        ctx.lineWidth = 1;
        ctx.beginPath();
        for(let i = 0; i < rect.width; i += 40) { ctx.moveTo(i, 0); ctx.lineTo(i, rect.height); }
        for(let i = 0; i < rect.height; i += 40) { ctx.moveTo(0, i); ctx.lineTo(rect.width, i); }
        ctx.stroke();
        
        ctx.strokeStyle = '#164024';
        ctx.beginPath(); ctx.moveTo(0, rect.height/2); ctx.lineTo(rect.width, rect.height/2); ctx.stroke();

        ctx.strokeStyle = waveColor;
        ctx.lineWidth = 2;
        ctx.shadowBlur = 8;
        ctx.shadowColor = waveColor;
        
        ctx.beginPath();
        for (let i = 0; i < dataBuffer.length; i++) {
            const x = i;
            const y = (rect.height / 2) - (dataBuffer[i] * (rect.height / 250));
            if (i === 0) ctx.moveTo(x, y);
            else ctx.lineTo(x, y);
        }
        ctx.stroke();
        ctx.shadowBlur = 0;
    }

    function runTest(isFault) {
        if (isTestActive || isEStopActive) return;
        
        const poles = document.getElementById('poles').value;
        const rating = document.getElementById('rating').value;

        isTestActive = true;
        currentTestType = isFault ? 'fault' : 'normal';
        waveColor = isFault ? '#ff3333' : '#00ff66';
        tick = 0;
        maxIp = 0;
        totalI2t = 0;
        
        setControls(true);
        
        const status = document.getElementById('status');
        status.className = 'status-bar testing';
        status.innerText = isFault ? 'INJECTING FAULT SCENARIO' : 'EXECUTING IEC TRIP PROTOCOL';
        
        logMessage(`Starting simulation: ${poles} MCB (${rating})`, 'new');
    }

    function finishTest() {
        isTestActive = false;
        setControls(false);
        const status = document.getElementById('status');
        
        if (currentTestType === 'normal') {
            status.className = 'status-bar pass';
            status.innerText = 'TEST COMPLETE : PASS';
            logMessage(`MCB Tripped. Peak: ${(maxIp/10).toFixed(2)}kA`, 'ok');
        } else {
            status.className = 'status-bar fail';
            status.innerText = 'BACKUP ISOLATION ENGAGED';
            logMessage('MCB Failed to trip. Safety contactor opened circuit.', 'err');
        }
    }

    function toggleEStop() {
        if (isEStopActive) {
            resetSystem();
        } else {
            engageEStop();
        }
    }

    function engageEStop() {
        isTestActive = false;
        isEStopActive = true;
        setControls(true);
        
        document.getElementById('btn-estop').disabled = false;
        
        waveColor = '#ff3333';
        
        for(let i = 0; i < dataBuffer.length; i++) dataBuffer[i] = 0;

        document.getElementById('main-container').classList.add('emergency-mode');
        const estopBtn = document.getElementById('btn-estop');
        estopBtn.innerText = '⚠ RESET SYSTEM ⚠';
        estopBtn.classList.add('btn-reset');

        const status = document.getElementById('status');
        status.className = 'status-bar fail';
        status.innerText = 'SYSTEM LOCKED: HARDWARE E-STOP ENGAGED';
        logMessage('CRITICAL: Operator triggered Emergency Stop. All relays opened.', 'err');
    }

    function resetSystem() {
        isEStopActive = false;
        setControls(false);
        waveColor = '#00ff66';

        document.getElementById('main-container').classList.remove('emergency-mode');
        const estopBtn = document.getElementById('btn-estop');
        estopBtn.innerText = 'E-STOP';
        estopBtn.classList.remove('btn-reset');

        const status = document.getElementById('status');
        status.className = 'status-bar';
        status.innerText = 'SYSTEM IDLE : READY FOR TEST';
        logMessage('Safety interlock reset. Relays energized and ready.', 'ok');
        
        document.getElementById('readout-ip').innerText = '0.00';
        document.getElementById('readout-i2t').innerText = '0.00';
    }
</script>
